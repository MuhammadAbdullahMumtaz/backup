USE master_erp;

-- ==========================================
-- Company Table
-- ==========================================
CREATE TABLE CompanyName (
    id INT AUTO_INCREMENT PRIMARY KEY,
    company_name VARCHAR(255) NOT NULL UNIQUE,

    add_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    update_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP
);

-- ==========================================
-- Logins Table
-- ==========================================
CREATE TABLE Logins (
    id INT AUTO_INCREMENT PRIMARY KEY,

    company_id INT NOT NULL,

    username VARCHAR(100) NOT NULL,
    password VARCHAR(255) NOT NULL,

    role ENUM('Admin', 'User')
    DEFAULT 'User',

    add_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    update_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT FK_Login_Company
        FOREIGN KEY (company_id)
        REFERENCES CompanyName(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE,

    UNIQUE(company_id, username)
);

-- ==========================================
-- Financial Year Table
-- ==========================================
CREATE TABLE FinancialYear (
    id INT AUTO_INCREMENT PRIMARY KEY,

    company_id INT NOT NULL,

    year_detail VARCHAR(50) NOT NULL,

    status ENUM('Active','Inactive')
    DEFAULT 'Inactive',

    current_month VARCHAR(25),

    start_date DATE NOT NULL,
    end_date DATE NOT NULL,

    add_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    update_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT FK_FinYear_Company
        FOREIGN KEY (company_id)
        REFERENCES CompanyName(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE,

    UNIQUE(company_id, year_detail, current_month)
);

-- BalanceSheet Table

    CONSTRAINT FK_BalanceSheet_Company
        FOREIGN KEY (company_id)

        ON DELETE CASCADE
        ON UPDATE CASCADE,

    UNIQUE(company_id, details)

-- ==========================================

CREATE TABLE VoucherType (
    id INT AUTO_INCREMENT PRIMARY KEY,

    company_id INT NOT NULL,

    type VARCHAR(10) NOT NULL,

    details VARCHAR(150) NOT NULL,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT fk_vouchertype_company
        FOREIGN KEY (company_id)
        REFERENCES CompanyName(id)
        ON DELETE RESTRICT
        ON UPDATE CASCADE,

    UNIQUE (company_id, details)
);

-- ==========================================
-- VoucherMain Table
-- ==========================================

CREATE TABLE VoucherMain (
    id INT AUTO_INCREMENT PRIMARY KEY,

    company_id INT NOT NULL,

    vouchertype_id INT NOT NULL,

    financialyear_id INT NOT NULL,

    voucher_no VARCHAR(30) NOT NULL,

    voucher_date DATE NOT NULL,

    voucher_month VARCHAR(25),

    isBalanced ENUM('Balanced', 'Unbalanced')
        NOT NULL DEFAULT 'Unbalanced',

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT fk_vouchermain_company
        FOREIGN KEY (company_id)
        REFERENCES CompanyName(id)
        ON DELETE RESTRICT
        ON UPDATE CASCADE,

    CONSTRAINT fk_vouchermain_vouchertype
        FOREIGN KEY (vouchertype_id)
        REFERENCES VoucherType(id)
        ON DELETE RESTRICT
        ON UPDATE CASCADE,

    CONSTRAINT fk_vouchermain_finyear
        FOREIGN KEY (financialyear_id)
        REFERENCES FinancialYear(id)
        ON DELETE RESTRICT
        ON UPDATE CASCADE,

    UNIQUE (company_id, financialyear_id, voucher_month, voucher_no)
);
